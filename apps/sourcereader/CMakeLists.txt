cmake_minimum_required(VERSION 3.12)
project(GetDAQSourceReader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(getdaq-source-ring-reader STATIC sourceringreader.cpp sourceringreadout.cpp sourcemfmring.cpp)
target_include_directories(getdaq-source-ring-reader PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(getdaq-source-ring-reader PUBLIC Threads::Threads)
target_compile_options(getdaq-source-ring-reader PRIVATE -Wall -Wextra -pedantic-errors)
include(CTest)
if(BUILD_TESTING)
  add_executable(source-ring-reader-tests ../../tests/source_ring_reader_test.cpp)
  target_link_libraries(source-ring-reader-tests PRIVATE getdaq-source-ring-reader)
  target_compile_options(source-ring-reader-tests PRIVATE -Wall -Wextra -pedantic-errors)
  add_test(NAME source-ring-reader COMMAND source-ring-reader-tests)
  add_executable(source-ring-readout-tests ../../tests/source_ring_readout_test.cpp)
  target_link_libraries(source-ring-readout-tests PRIVATE getdaq-source-ring-reader)
  target_compile_options(source-ring-readout-tests PRIVATE -Wall -Wextra -pedantic-errors)
  add_test(NAME source-ring-readout COMMAND source-ring-readout-tests)
  add_executable(source-mfm-ring-tests ../../tests/source_mfm_ring_test.cpp)
  target_link_libraries(source-mfm-ring-tests PRIVATE getdaq-source-ring-reader)
  target_compile_options(source-mfm-ring-tests PRIVATE -Wall -Wextra -pedantic-errors)
  add_test(NAME source-mfm-ring COMMAND source-mfm-ring-tests)
endif()
