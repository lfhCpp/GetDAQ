cmake_minimum_required(VERSION 3.12)
project(GetDAQEccCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(getdaq-ecc-core STATIC eccstatemachine.cpp)
target_include_directories(getdaq-ecc-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(getdaq-ecc-core PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(getdaq-ecc-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
option(GETDAQ_ECC_JOURNAL "Build the platform-native persistent scan journal" ON)
if(GETDAQ_ECC_JOURNAL)
  if(NOT TARGET getdaq-experiment-core)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../common/experiment" "${CMAKE_CURRENT_BINARY_DIR}/experiment")
  endif()
  if(NOT TARGET getdaq-control-protocol)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../common/control" "${CMAKE_CURRENT_BINARY_DIR}/control")
  endif()
endif()
if(GETDAQ_ECC_JOURNAL AND UNIX)
  add_library(getdaq-ecc-journal STATIC eccjournal.cpp eccjournal_replay.cpp eccjournal_codec.cpp
    scanexecutionjournalstore.cpp scanexecutionjournalstore_read.cpp scanexecutionjournalstore_publish.cpp
    scanruntimeidentitybinding.cpp scanruntimeidentitycoordinator.cpp
    scaneventcountbaselinecoordinator.cpp scanoperationrunner.cpp)
  target_link_libraries(getdaq-ecc-journal PUBLIC getdaq-ecc-core
    getdaq-experiment-core getdaq-control-protocol)
  add_executable(getdaq-ecc-journal-tool journalmain.cpp)
  target_link_libraries(getdaq-ecc-journal-tool PRIVATE getdaq-ecc-journal)
  set_target_properties(getdaq-ecc-journal-tool PROPERTIES OUTPUT_NAME getdaq-ecc-journal RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  if(BUILD_TESTING)
    add_executable(ecc-journal-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ecc_journal_test.cpp")
    target_link_libraries(ecc-journal-tests PRIVATE getdaq-ecc-journal)
    add_dependencies(ecc-journal-tests getdaq-ecc-journal-tool)
    target_compile_definitions(ecc-journal-tests PRIVATE
      GETDAQ_ECC_JOURNAL_TOOL="$<TARGET_FILE:getdaq-ecc-journal-tool>")
    set_target_properties(ecc-journal-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    add_test(NAME ecc-journal COMMAND ecc-journal-tests)
    set_tests_properties(ecc-journal PROPERTIES TIMEOUT 20)
  endif()
elseif(GETDAQ_ECC_JOURNAL AND WIN32)
  add_library(getdaq-scan-journal STATIC scanexecutionjournalstore_windows.cpp
    scanexecutionjournalstore_windows_io.cpp scanexecutionjournalstore_windows_support.cpp
    scanruntimeidentitybinding.cpp scanruntimeidentitycoordinator.cpp
    scaneventcountbaselinecoordinator.cpp scanoperationrunner.cpp)
  target_link_libraries(getdaq-scan-journal PUBLIC getdaq-ecc-core
    getdaq-experiment-core getdaq-control-protocol PRIVATE advapi32)
  add_executable(getdaq-scan-journal-tool scanjournalmain.cpp)
  target_link_libraries(getdaq-scan-journal-tool PRIVATE getdaq-scan-journal shell32)
  set_target_properties(getdaq-scan-journal-tool PROPERTIES OUTPUT_NAME getdaq-scan-journal
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  if(BUILD_TESTING)
    add_executable(scan-journal-store-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/scanexecutionjournalstore_windows_test.cpp")
    target_link_libraries(scan-journal-store-tests PRIVATE getdaq-scan-journal advapi32)
    add_test(NAME scan-journal-store COMMAND scan-journal-store-tests)
    set_tests_properties(scan-journal-store PROPERTIES TIMEOUT 20)
  endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  if(TARGET getdaq-ecc-journal)
    target_compile_options(getdaq-ecc-journal PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(TARGET getdaq-scan-journal)
    target_compile_options(getdaq-scan-journal PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(ecc-state-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ecc_state_test.cpp")
  target_link_libraries(ecc-state-tests PRIVATE getdaq-ecc-core)
  add_test(NAME ecc-state COMMAND ecc-state-tests)
  set_tests_properties(ecc-state PROPERTIES TIMEOUT 15)
endif()
