cmake_minimum_required(VERSION 3.12)
project(GetDAQConfigurationRepository LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT UNIX)
  message(FATAL_ERROR "The server repository backend currently requires POSIX; Qt clients remain separate")
endif()
if(NOT TARGET getdaq-experiment-core)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../common/experiment" "${CMAKE_BINARY_DIR}/experiment")
endif()
find_package(Threads REQUIRED)
add_library(getdaq-configuration-repository STATIC configurationrepository.cpp configurationrepository_files.cpp configurationrepository_plan.cpp repositorycredential.cpp)
target_include_directories(getdaq-configuration-repository PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(getdaq-configuration-repository PUBLIC getdaq-experiment-core Threads::Threads)
target_compile_options(getdaq-configuration-repository PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
add_executable(getdaq-config-repository main.cpp)
target_link_libraries(getdaq-config-repository PRIVATE getdaq-configuration-repository)
add_executable(configuration-repository-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/configuration_repository_core_test.cpp")
target_link_libraries(configuration-repository-tests PRIVATE getdaq-configuration-repository)
target_include_directories(configuration-repository-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../tests")
foreach(program getdaq-config-repository configuration-repository-tests)
  target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endforeach()
if(BUILD_TESTING)
  add_test(NAME configuration-repository-core COMMAND configuration-repository-tests --self-contained-core)
  set_tests_properties(configuration-repository-core PROPERTIES TIMEOUT 180)
endif()
find_package(Qt5 5.12 COMPONENTS Core Network Concurrent QUIET)
if(Qt5Core_FOUND AND Qt5Network_FOUND AND Qt5Concurrent_FOUND)
  add_executable(getdaq-config-service servicemain.cpp repositoryservice.cpp repositoryservice.h
    "${CMAKE_CURRENT_SOURCE_DIR}/../../common/http/httpserver.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../common/http/httpserver.h")
  target_include_directories(getdaq-config-service PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../common/http")
  target_link_libraries(getdaq-config-service PRIVATE getdaq-configuration-repository Qt5::Core Qt5::Network Qt5::Concurrent)
  target_compile_options(getdaq-config-service PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
  set_target_properties(getdaq-config-service PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
endif()
