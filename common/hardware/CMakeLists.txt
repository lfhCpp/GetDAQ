cmake_minimum_required(VERSION 3.12)
project(GetDAQHardwareDescription LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET getdaq-experiment-core)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../experiment" "${CMAKE_CURRENT_BINARY_DIR}/experiment")
endif()
add_library(getdaq-hardware-description-core STATIC hardwaredescription.cpp)
target_include_directories(getdaq-hardware-description-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(getdaq-hardware-description-core PUBLIC getdaq-experiment-core)
target_compile_options(getdaq-hardware-description-core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(hardware-description-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/hardware_description_core_test.cpp")
  target_link_libraries(hardware-description-tests PRIVATE getdaq-hardware-description-core)
  add_test(NAME hardware-description COMMAND hardware-description-tests)
endif()
