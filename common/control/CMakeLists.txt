cmake_minimum_required(VERSION 3.12)
project(GetDAQControlProtocol LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET getdaq-experiment-core)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../experiment" "${CMAKE_CURRENT_BINARY_DIR}/experiment")
endif()
add_library(getdaq-control-protocol STATIC eccprotocol.cpp eccprotocol_request.cpp eccprotocol_request_validation.cpp eccprotocol_configuration.cpp eccprotocol_observation.cpp eccplantransitionprotocol.cpp transferprotocol.cpp hardwarestateprotocol.cpp coldhandoffprotocol.cpp scanruntimeidentity.cpp scanruntimeidentity_initial.cpp scanruntimeidentity_codec.cpp scanruntimeidentity_deployment.cpp scaneventcountobservation.cpp)
target_include_directories(getdaq-control-protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(getdaq-control-protocol PUBLIC getdaq-experiment-core)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(getdaq-control-protocol PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ecc-protocol-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ecc_protocol_test.cpp")
  target_link_libraries(ecc-protocol-tests PRIVATE getdaq-control-protocol)
  add_test(NAME ecc-protocol COMMAND ecc-protocol-tests)
  add_executable(cold-handoff-protocol-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/cold_handoff_protocol_test.cpp")
  target_link_libraries(cold-handoff-protocol-tests PRIVATE getdaq-control-protocol)
  add_test(NAME cold-handoff-protocol COMMAND cold-handoff-protocol-tests)
  add_executable(scan-runtime-identity-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/scan_runtime_identity_test.cpp")
  target_link_libraries(scan-runtime-identity-tests PRIVATE getdaq-control-protocol)
  add_test(NAME scan-runtime-identity COMMAND scan-runtime-identity-tests)
  add_executable(scan-event-count-observation-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/scan_event_count_observation_test.cpp")
  target_link_libraries(scan-event-count-observation-tests PRIVATE getdaq-control-protocol)
  add_test(NAME scan-event-count-observation COMMAND scan-event-count-observation-tests)
endif()
