cmake_minimum_required(VERSION 3.12)
project(GetDAQConfigurationCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(getdaq-configuration-range STATIC configurationrange.cpp configurationrange_parse.cpp)
target_include_directories(getdaq-configuration-range PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(getdaq-configuration-range PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(configuration-range-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/configuration_range_test.cpp")
  target_link_libraries(configuration-range-tests PRIVATE getdaq-configuration-range)
  add_test(NAME configuration-range COMMAND configuration-range-tests)
endif()
